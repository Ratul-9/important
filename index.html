/* Q2. Count characters, words, lines, spaces and tabs in input.c
       and write the result to output.txt */
#include <stdio.h>
#include <ctype.h>

int main()
{
    FILE *in = fopen("input.c", "r");
    FILE *out = fopen("output.txt", "w");
    int ch, chars = 0, words = 0, lines = 0, spaces = 0, tabs = 0;
    int inWord = 0;

    if (in == NULL) { printf("Cannot open file\n"); return 1; }

    while ((ch = fgetc(in)) != EOF) {
        chars++;
        if (ch == '\n') lines++;
        if (ch == ' ')  spaces++;
        if (ch == '\t') tabs++;

        if (isspace(ch))
            inWord = 0;              /* word ended          */
        else if (inWord == 0) {
            inWord = 1;              /* a new word started  */
            words++;
        }
    }

    fprintf(out, "Characters : %d\n", chars);
    fprintf(out, "Words      : %d\n", words);
    fprintf(out, "Lines      : %d\n", lines);
    fprintf(out, "Spaces     : %d\n", spaces);
    fprintf(out, "Tabs       : %d\n", tabs);

    fclose(in);
    fclose(out);
    return 0;
}

Q3 – Remove all single-line (//) and multi-line (/* */) comments from input.c and write the program to output.txt. ★

Code – q03_remove_comments.c

/* Q3. Remove all comments ( // ... and slash-star ... star-slash )
       from input.c and write the clean program to output.txt.
       Comments inside string literals must NOT be touched. */
#include <stdio.h>

int main()
{
    FILE *in = fopen("input.c", "r");
    FILE *out = fopen("output.txt", "w");
    int ch, next, count = 0;

    if (in == NULL) { printf("Cannot open file\n"); return 1; }

    while ((ch = fgetc(in)) != EOF) {
        if (ch == '"') {                       /* copy string as it is */
            fputc(ch, out);
            while ((ch = fgetc(in)) != EOF && ch != '"') {
                fputc(ch, out);
                if (ch == '\\') fputc(fgetc(in), out);   /* \" inside string */
            }
            fputc('"', out);
        }
        else if (ch == '/') {
            next = fgetc(in);
            if (next == '/') {                 /* single-line comment */
                while ((ch = fgetc(in)) != EOF && ch != '\n');
                fputc('\n', out);
                count++;
            }
            else if (next == '*') {            /* multi-line comment */
                int prev = 0;
                while ((ch = fgetc(in)) != EOF) {
                    if (prev == '*' && ch == '/') break;
                    prev = ch;
                }
                count++;
            }
            else {                             /* just a division sign */
                fputc(ch, out);
                if (next != EOF) ungetc(next, in);
            }
        }
        else
            fputc(ch, out);
    }
    printf("Comments removed: %d\n", count);
    fclose(in);
    fclose(out);
    return 0;
}

Q4 – Remove extra spaces, tabs and blank lines from input.c (many spaces become one space).

Code – q04_whitespace.c

/* Q4. Remove extra white spaces, tabs and blank lines from input.c.
       Multiple spaces/tabs -> one space, leading spaces removed,
       empty lines removed. Output in output.txt */
#include <stdio.h>
#include <string.h>

int main()
{
    FILE *in = fopen("input.c", "r");
    FILE *out = fopen("output.txt", "w");
    char line[500], clean[500];
    int i, k;

    if (in == NULL) { printf("Cannot open file\n"); return 1; }

    while (fgets(line, sizeof(line), in) != NULL) {   /* read one line */
        k = 0;
        for (i = 0; line[i] != '\0' && line[i] != '\n'; i++) {
            if (line[i] == ' ' || line[i] == '\t') {
                /* add one space only if previous char is not a space
                   and we are not at the beginning of the line          */
                if (k > 0 && clean[k - 1] != ' ')
                    clean[k++] = ' ';
            } else
                clean[k++] = line[i];
        }
        if (k > 0 && clean[k - 1] == ' ') k--;   /* trailing space */
        clean[k] = '\0';

        if (strlen(clean) > 0)                   /* skip blank lines */
            fprintf(out, "%s\n", clean);
    }
    fclose(in);
    fclose(out);
    return 0;
}

Code – q05_lexer.c

/* Q5. LEXICAL ANALYZER in C.
       Read input.c, break it into tokens and write every token with its
       type and line number to output.txt. Comments are skipped.
       Token types: KEYWORD, IDENTIFIER, CONSTANT, STRING, CHAR CONST,
                    OPERATOR, SPECIAL SYMBOL, PREPROCESSOR            */
#include <stdio.h>
#include <string.h>
#include <ctype.h>

char *keywords[] = {"auto","break","case","char","const","continue",
    "default","do","double","else","enum","extern","float","for","goto",
    "if","int","long","register","return","short","signed","sizeof",
    "static","struct","switch","typedef","union","unsigned","void",
    "volatile","while"};

char *ops2[] = {"++","--","==","!=","<=",">=","&&","||",
                "+=","-=","*=","/=","%=","<<",">>","->"};

int isKeyword(char *w)
{
    int i;
    for (i = 0; i < 32; i++)
        if (strcmp(w, keywords[i]) == 0) return 1;
    return 0;
}

int isOp2(char *p)                 /* is p[0]p[1] a 2-char operator? */
{
    int i;
    for (i = 0; i < 16; i++)
        if (strncmp(p, ops2[i], 2) == 0) return 1;
    return 0;
}

int main()
{
    FILE *in = fopen("input.c", "r");
    FILE *out = fopen("output.txt", "w");
    char s[20000], tok[200];
    int n = 0, ch, i = 0, k, line = 1;
    int nk = 0, ni = 0, nc = 0, ns = 0, no = 0, nsp = 0;

    if (in == NULL) { printf("Cannot open file\n"); return 1; }
    while ((ch = fgetc(in)) != EOF) s[n++] = ch;   /* whole file -> s */
    s[n] = '\0';

    fprintf(out, "%-6s%-26s%s\n", "LINE", "LEXEME", "TOKEN");
    while (s[i] != '\0') {
        k = 0;
        if (s[i] == '\n') { line++; i++; continue; }
        if (isspace(s[i])) { i++; continue; }

        if (s[i] == '/' && s[i+1] == '/') {            /* // comment */
            while (s[i] && s[i] != '\n') i++;
            continue;
        }
        if (s[i] == '/' && s[i+1] == '*') {            /* block comment */
            i += 2;
            while (s[i] && !(s[i] == '*' && s[i+1] == '/')) {
                if (s[i] == '\n') line++;
                i++;
            }
            if (s[i]) i += 2;
            continue;
        }
        if (s[i] == '#') {                             /* #include ... */
            while (s[i] && s[i] != '\n') tok[k++] = s[i++];
            tok[k] = '\0';
            fprintf(out, "%-6d%-26s%s\n", line, tok, "PREPROCESSOR");
            continue;
        }
        if (isalpha(s[i]) || s[i] == '_') {            /* word */
            while (isalnum(s[i]) || s[i] == '_') tok[k++] = s[i++];
            tok[k] = '\0';
            if (isKeyword(tok)) { fprintf(out, "%-6d%-26s%s\n", line, tok, "KEYWORD"); nk++; }
            else                { fprintf(out, "%-6d%-26s%s\n", line, tok, "IDENTIFIER"); ni++; }
        }
        else if (isdigit(s[i])) {                      /* number */
            while (isdigit(s[i]) || s[i] == '.') tok[k++] = s[i++];
            tok[k] = '\0';
            fprintf(out, "%-6d%-26s%s\n", line, tok, "CONSTANT"); nc++;
        }
        else if (s[i] == '"' || s[i] == '\'') {        /* "string" / 'c' */
            char q = s[i];
            tok[k++] = s[i++];
            while (s[i] && s[i] != q) {
                if (s[i] == '\\') tok[k++] = s[i++];   /* escape: \n \" */
                tok[k++] = s[i++];
            }
            tok[k++] = s[i++];
            tok[k] = '\0';
            fprintf(out, "%-6d%-26s%s\n", line, tok, q == '"' ? "STRING" : "CHAR CONST"); ns++;
        }
        else if (strchr("+-*/%=<>!&|^~", s[i])) {      /* operator */
            if (isOp2(&s[i])) { tok[k++] = s[i++]; }
            tok[k++] = s[i++];
            tok[k] = '\0';
            fprintf(out, "%-6d%-26s%s\n", line, tok, "OPERATOR"); no++;
        }
        else {                                         /* ( ) { } [ ] ; , */
            tok[0] = s[i++]; tok[1] = '\0';
            fprintf(out, "%-6d%-26s%s\n", line, tok, "SPECIAL SYMBOL"); nsp++;
        }
    }
    fprintf(out, "\nKeywords=%d Identifiers=%d Constants=%d Strings=%d "
                 "Operators=%d Special=%d\nTOTAL TOKENS=%d\n",
            nk, ni, nc, ns, no, nsp, nk + ni + nc + ns + no + nsp);
    fclose(in);
    fclose(out);
    return 0;
}

/* Q6. *** MOST EXPECTED ***
   Read a C program from input.c. For every for loop, print all the
   tokens between its opening '(' and closing ')' into output.txt.
   e.g.  for(i = 0; i < n; i++)  ->  i  =  0  ;  i  <  n  ;  i  ++
   ("for" inside strings/comments is ignored; nested ( ) handled.)
   To do the same for while / if / switch just change KEY below.   */
#include <stdio.h>
#include <string.h>
#include <ctype.h>

char KEY[] = "for";

char *keywords[] = {"auto","break","case","char","const","continue",
    "default","do","double","else","enum","extern","float","for","goto",
    "if","int","long","register","return","short","signed","sizeof",
    "static","struct","switch","typedef","union","unsigned","void",
    "volatile","while"};

int isKeyword(char *w)
{
    int i;
    for (i = 0; i < 32; i++)
        if (strcmp(w, keywords[i]) == 0) return 1;
    return 0;
}

/* tokenize s[start .. end-1] and write each token with its type */
int printTokens(FILE *out, char *s, int start, int end)
{
    char tok[100];
    int i = start, k, count = 0;
    char *type;

    while (i < end) {
        k = 0;
        if (isspace(s[i])) { i++; continue; }

        if (isalpha(s[i]) || s[i] == '_') {
            while (i < end && (isalnum(s[i]) || s[i] == '_')) tok[k++] = s[i++];
            tok[k] = '\0';
            type = isKeyword(tok) ? "KEYWORD" : "IDENTIFIER";
        }
        else if (isdigit(s[i])) {
            while (i < end && (isdigit(s[i]) || s[i] == '.')) tok[k++] = s[i++];
            tok[k] = '\0';
            type = "CONSTANT";
        }
        else if (s[i] == '"') {
            tok[k++] = s[i++];
            while (i < end && s[i] != '"') tok[k++] = s[i++];
            tok[k++] = s[i++];
            tok[k] = '\0';
            type = "STRING";
        }
        else if (strchr("+-*/%=<>!&|", s[i])) {
            tok[k++] = s[i++];
            /* 2-char operators: ++ -- && || == != <= >= += -= *= /= */
            if (s[i] == '=' || (s[i] == tok[0] && strchr("+-&|<>", s[i])))
                tok[k++] = s[i++];
            tok[k] = '\0';
            type = "OPERATOR";
        }
        else {
            tok[0] = s[i++]; tok[1] = '\0';
            type = (tok[0] == ';' || tok[0] == ',') ? "SEPARATOR" : "SPECIAL SYMBOL";
        }
        fprintf(out, "   %-10s %s\n", tok, type);
        count++;
    }
    return count;
}

/* simple tokenizer: prints every token of s[start..end-1] on one line */
int printTokens(FILE *out, char *s, int start, int end)
{
    int i = start, count = 0;
    while (i < end) {
        if (isspace(s[i])) { i++; continue; }
        fprintf(out, "   ");
        if (isalnum(s[i]) || s[i] == '_') {                 /* word/number */
            while (i < end && (isalnum(s[i]) || s[i] == '_' || s[i] == '.'))
                fputc(s[i++], out);
        }
        else if (s[i] == '"') {                             /* string */
            fputc(s[i++], out);
            while (i < end && s[i] != '"') {
                if (s[i] == '\\') fputc(s[i++], out);
                fputc(s[i++], out);
            }
            fputc(s[i++], out);
        }
        else if (strchr("+-*/%=<>!&|", s[i]) &&
                 (s[i+1] == '=' || (s[i+1] == s[i] && strchr("+-&|", s[i])))) {
            fputc(s[i++], out); fputc(s[i++], out);         /* ++ == && */
        }
        else fputc(s[i++], out);                            /* 1 char  */
        fputc('\n', out);
        count++;
    }
    return count;
}

/* Q9. Check whether all brackets ( ) { } [ ] in input.c are balanced,
       using a STACK. Report the line number of the first error.
       Brackets inside strings, char constants and comments are ignored. */
#include <stdio.h>

char stk[1000];
int  stkLine[1000];
int  top = -1;

int main()
{
    FILE *in = fopen("input.c", "r");
    FILE *out = fopen("output.txt", "w");
    int ch, next, line = 1, error = 0;

    if (in == NULL) { printf("Cannot open input.c\n"); return 1; }

    while ((ch = fgetc(in)) != EOF && !error) {
        if (ch == '\n') line++;
        else if (ch == '"' || ch == '\'') {            /* skip "..." '..' */
            int q = ch;
            while ((ch = fgetc(in)) != EOF && ch != q)
                if (ch == '\\') fgetc(in);
        }
        else if (ch == '/') {
            next = fgetc(in);
            if (next == '/') { while ((ch = fgetc(in)) != EOF && ch != '\n'); line++; }
            else if (next == '*') {
                int prev = 0;
                while ((ch = fgetc(in)) != EOF) {
                    if (ch == '\n') line++;
                    if (prev == '*' && ch == '/') break;
                    prev = ch;
                }
            }
            else ungetc(next, in);
        }
        else if (ch == '(' || ch == '{' || ch == '[') {   /* PUSH */
            stk[++top] = ch;
            stkLine[top] = line;
        }
        else if (ch == ')' || ch == '}' || ch == ']') {   /* POP & match */
            char want = (ch == ')') ? '(' : (ch == '}') ? '{' : '[';
            if (top == -1) {
                fprintf(out, "Error: extra '%c' at line %d\n", ch, line);
                error = 1;
            } else if (stk[top] != want) {
                fprintf(out, "Error: '%c' at line %d does not match '%c' opened at line %d\n",
                        ch, line, stk[top], stkLine[top]);
                error = 1;
            } else
                top--;
        }
    }
    if (!error && top != -1)
        fprintf(out, "Error: '%c' opened at line %d is never closed\n", stk[top], stkLine[top]);
    else if (!error)
        fprintf(out, "All brackets are BALANCED\n");

    fclose(in);
    fclose(out);
    return 0;
}

/* Q23. Count LINES OF CODE in input.c:
          blank lines, comment lines, preprocessor lines and code lines.
        (A line that is only a comment, or inside a block comment,
         counts as a comment line.)                         -> output.txt */
#include <stdio.h>
#include <string.h>

int main()
{
    FILE *in = fopen("input.c", "r");
    FILE *out = fopen("output.txt", "w");
    char line[500];
    int blank = 0, comment = 0, pre = 0, code = 0, total = 0;
    int inBlock = 0;                          /* inside a block comment? */

    if (in == NULL) { printf("Cannot open input.c\n"); return 1; }

    while (fgets(line, sizeof(line), in)) {
        char *p = line;
        total++;
        while (*p == ' ' || *p == '\t') p++;  /* skip leading spaces */

        if (inBlock) {
            comment++;
            if (strstr(p, "*/")) inBlock = 0;
        }
        else if (*p == '\n' || *p == '\0') blank++;
        else if (strncmp(p, "//", 2) == 0) comment++;
        else if (strncmp(p, "/*", 2) == 0) {
            comment++;
            if (strstr(p + 2, "*/") == NULL) inBlock = 1;   /* continues */
        }
        else if (*p == '#') pre++;
        else code++;
    }
    fprintf(out, "Total lines        : %d\n", total);
    fprintf(out, "Blank lines        : %d\n", blank);
    fprintf(out, "Comment lines      : %d\n", comment);
    fprintf(out, "Preprocessor lines : %d\n", pre);
    fprintf(out, "Code lines         : %d\n", code);
    fclose(in);
    fclose(out);
    return 0;
}

LEX – structure, regular expressions, variables, commands

WHAT IS LEX?

Lex (on Ubuntu: flex) is a LEXICAL ANALYZER GENERATOR. You write patterns
(regular expressions) plus C actions in a .l file. Lex turns it into a C file,
lex.yy.c, which contains the function yylex().

    file.l  --lex-->  lex.yy.c  --gcc-->  a.out  --> tokens / output

STRUCTURE OF A .l FILE (3 sections, separated by %%)

%{
/* 1. DEFINITIONS: C code copied as it is (headers, globals) */
#include <stdio.h>
int count = 0;
%}
DIGIT   [0-9]
ID      [a-zA-Z_][a-zA-Z0-9_]*
%%
{DIGIT}+       { printf("NUMBER %s\n", yytext); }
{ID}            { count++; }
\n              ;                   /* empty action = ignore */
.               ;
%%
/* 3. USER CODE: main() and helper functions */
int yywrap() { return 1; }
int main() { yylex(); printf("%d\n", count); return 0; }
  Section 1 = definitions (%{ C code %} + named patterns)
  Section 2 = rules       pattern   { action }
  Section 3 = user code   main(), yywrap(), helpers
➜ Patterns must start in COLUMN 1 (no space before them in the rules section).
➜ Put C comments in the rules section AFTER the action, or indented. Never at column 1.

REGULAR EXPRESSIONS IN LEX

  x        the char x               "abc"    the exact string abc
  .        any char EXCEPT \n       \n \t    newline, tab
  [abc]    a or b or c              [a-z]    range
  [^abc]   any char except a b c    [^\n]    any char but newline
  r*       0 or more                r+       1 or more
  r?       0 or 1 (optional)        r{2,5}   2 to 5 times   r{3} exactly 3
  r|s      r or s                   (r)      grouping
  ^r       r at start of a line     r$       r at end of a line
  r/s      r only if followed by s (trailing context)
  \.  \*  \+  \(   escape special chars,  or quote them: "." "*" "+" "("
  {NAME}   use a definition

SPECIAL VARIABLES & FUNCTIONS

  yytext     char*  the text that matched
  yyleng     int    length of yytext
  yylex()    the scanner. Call it to start; returns a token (with yacc)
  yywrap()   called at EOF. return 1 = stop, 0 = more input
  yyin       FILE*  input  (default stdin)  -> yyin = fopen("input.c","r");
  yyout      FILE*  output (default stdout) -> yyout = fopen("out.txt","w");
  ECHO       copy yytext to yyout
  yylval     value passed to yacc together with the token
  BEGIN S    switch to start condition S;  BEGIN INITIAL = back to normal
  %x S       exclusive start condition (only <S> rules are active)
  %s S       inclusive start condition (<S> rules + normal rules)
  yymore()   append the next match to yytext
  yyless(n)  keep only the first n chars, push the rest back
  input()    read 1 char,   unput(c)  push 1 char back

HOW LEX CHOOSES A RULE (very common viva question)

  1. LONGEST MATCH wins:  "<=" beats "<",  identifier "format" beats keyword "for"
  2. If two rules match the SAME length, the rule written FIRST wins.
     -> write keyword rules ABOVE the identifier rule!

COMMON PATTERNS

[0-9]+                          integer
[0-9]+"."[0-9]+                 real number
[+-]?[0-9]+                     signed integer
[a-zA-Z_][a-zA-Z0-9_]*          identifier
\"(\\.|[^"\\])*\"               string literal "..."
"//".*                          single line comment
"/*"([^*]|\*+[^*/])*\*+"/"      multi line comment
[ \t]+                          white space
^[ \t]*\n                       blank line

COMPILE & RUN

lex prog.l                 # or: flex prog.l   -> creates lex.yy.c
gcc lex.yy.c -o prog       # works because we wrote yywrap() ourselves
gcc lex.yy.c -o prog -ll   # if yywrap() is NOT defined (or use -lfl)
./prog                     # type input, end with Ctrl+D
./prog < input.txt         # input from a file via the shell

ERRORS YOU MAY SEE

  undefined reference to `yywrap'  -> add int yywrap(){return 1;} or link with -ll
  unrecognized rule                -> space before the pattern / bad quoting
  premature EOF                    -> missing %% or unclosed %{ / " / [

YACC – structure, keywords, Lex + Yacc commands

WHAT IS YACC?

Yacc (Yet Another Compiler Compiler; on Ubuntu: bison -y) is a PARSER GENERATOR.
You give it a context-free GRAMMAR with C actions. It produces an LALR(1)
parser, yyparse(), in y.tab.c. Lex gives the tokens; yacc checks the grammar.

  input --> yylex() [lex.yy.c] --tokens+yylval--> yyparse() [y.tab.c] --> actions

STRUCTURE OF A .y FILE

%{
#include <stdio.h>
#include <stdlib.h>
int yylex(void);                    /* prototypes: avoid gcc warnings */
void yyerror(const char *s);
%}
%token NUMBER ID                    /* terminals that lex returns     */
%left '+' '-'                       /* LOW precedence                 */
%left '*' '/'                       /* higher (written later)         */
%right UMINUS                       /* highest                        */
%%
line : expr '\n'          { printf("= %d\n", $1); }
     ;
expr : expr '+' expr      { $$ = $1 + $3; }
     | expr '*' expr      { $$ = $1 * $3; }
     | '(' expr ')'       { $$ = $2; }
     | '-' expr %prec UMINUS { $$ = -$2; }
     | NUMBER             { $$ = $1; }
     ;
%%
void yyerror(const char *s) { printf("Error: %s\n", s); }
int main() { yyparse(); return 0; }

MATCHING LEX FILE

%{
#include <stdlib.h>
#include "y.tab.h"          /* token numbers made by  yacc -d  */
%}
%%
[0-9]+    { yylval = atoi(yytext); return NUMBER; }
[ \t]     ;
\n        { return '\n'; }
.         { return yytext[0]; }   /* + - * / ( ) as char tokens */
%%
int yywrap() { return 1; }

YACC KEYWORDS

  %token A B          declare tokens (terminals)
  %left / %right      operator associativity + precedence (later line = higher)
  %nonassoc           no associativity (a < b < c is an error)
  %prec X             give this rule the precedence of X (unary minus)
  %start s            start symbol (default: left side of the first rule)
  %union {int i; char *s;}  several value types
  %token <s> ID       token with the type s from the union
  %type  <i> expr     non-terminal with a type
  $$                  value of the left-hand side
  $1 $2 $3            values of the 1st, 2nd, 3rd symbols on the right
  yylval              value set by lex for the current token
  yyparse()           start parsing: returns 0 = success, 1 = error
  yyerror(s)          called automatically on a syntax error
  error / yyerrok     error-recovery token / resume after an error
  YYSTYPE             type of values (default int); #define YYSTYPE double

COMPILE & RUN (ORDER MATTERS)

lex  calc.l                       # -> lex.yy.c
yacc -d calc.y                    # -> y.tab.c + y.tab.h  (-d makes the .h!)
gcc lex.yy.c y.tab.c -o calc      # add -ll if yywrap() is not defined
./calc
  (bison alternative:  bison -dy calc.y  gives the same file names)

ERRORS YOU MAY SEE

  y.tab.h: No such file        -> you forgot -d in  yacc -d
  conflicts: 1 shift/reduce    -> ambiguous grammar; add %left / %right
  syntax error at runtime      -> input does not match the grammar (yyerror)
  implicit declaration of yylex-> add  int yylex(void);  in the .y file

GRAMMAR TIPS

➜ Prefer LEFT recursion: list : list ',' ID | ID ;   (uses less stack in LALR)
➜ Empty rule:   opt : /* empty */ | X ;
➜ Char tokens like '+' need no %token; lex just does  return yytext[0];

Viva Questions & Answers

Q1. What is a compiler? Name its phases.

A compiler translates a high-level program into target code. Phases: Lexical analysis -> Syntax analysis -> Semantic analysis -> Intermediate code generation -> Code optimization -> Code generation. The symbol table manager and the error handler work across all phases.

Q2. What does the lexical analyzer do?

It reads characters and groups them into TOKENS (keywords, identifiers, constants, operators, symbols). It removes white space and comments, keeps line numbers for error messages, and enters identifiers into the symbol table.

Q3. Token vs lexeme vs pattern?

Token = category (IDENTIFIER). Lexeme = the actual text (sum). Pattern = the rule that describes the lexemes of a token ([a-zA-Z_][a-zA-Z0-9_]*).

Q4. How many tokens are in for(i=0;i<n;i++) ?

for ( i = 0 ; i < n ; i ++ )  = 13 tokens (counting the brackets). Between the brackets there are 10.

Q5. What is a symbol table?

A data structure storing information about identifiers: name, type, scope, size, line and address. It is built in the lexical/syntax phases and used by every later phase.

Q6. What is Lex? What is Yacc?

Lex is a lexical analyzer generator: regular expressions in, lex.yy.c with yylex() out. Yacc is a parser generator: a CFG in, y.tab.c with yyparse() out. It is an LALR(1) parser.

Q7. What are yytext, yyleng, yyin, yyout?

yytext = the matched lexeme, yyleng = its length, yyin = the input FILE* (default stdin), yyout = the output FILE* (default stdout).

Q8. What is yywrap()?

It is called by yylex() at end of input. Return 1 means no more input, so stop. Return 0 means more input is ready (e.g. another file in yyin). Without it you must link with -ll or -lfl.

Q9. What happens when two lex rules match?

The longest match is chosen. If the lengths are equal, the rule that appears first in the file is chosen. That is why keyword rules come before the identifier rule.

Q10. What is a start condition in lex?

A state that turns groups of rules on or off. %x declares an exclusive state and %s an inclusive one. BEGIN STATE enters it and BEGIN INITIAL returns to normal. It is used for comments, strings, or tokens inside for(...).

Q11. What is yylval?

A global variable through which lex passes the VALUE of a token to yacc (e.g. the number 25 for the token NUMBER). Its type is YYSTYPE (int by default).

Q12. What do $$ and $1 mean in yacc?

$$ = the value of the non-terminal on the left side. $1, $2 ... = the values of the 1st, 2nd ... symbols on the right side of the rule.

Q13. What does yacc -d do?

It also creates y.tab.h, which holds the #define token numbers and the yylval declaration. The lex file includes it so both agree on the token codes.

Q14. Order of commands for lex + yacc?

lex file.l ; yacc -d file.y ; gcc lex.yy.c y.tab.c -o out ; ./out. Run yacc -d before compiling lex.yy.c because it needs y.tab.h.

Q15. What is a shift/reduce conflict?

The parser cannot decide whether to shift the next token or reduce by a rule. It happens with ambiguous grammars like E -> E+E. Fix it with %left/%right precedence or by rewriting the grammar. By default yacc shifts.

Q16. What is a reduce/reduce conflict?

Two different rules can reduce the same input. Yacc picks the rule listed first. It usually means the grammar is wrong.

Q17. What is %left / %right / %nonassoc?

They declare the associativity of operators. Their order sets precedence: a line written later has higher precedence. %nonassoc forbids chaining (a<b<c).

Q18. What is %prec?

It gives a rule the precedence of a named token, e.g. '-' expr %prec UMINUS makes unary minus bind tighter than * and /.

Q19. Which parser does yacc build?

An LALR(1) bottom-up shift-reduce parser: Look-Ahead LR with 1 token of lookahead.

Q20. Top-down vs bottom-up parsing?

Top-down (LL, recursive descent) starts from the start symbol and expands it to the input. Bottom-up (LR, SLR, LALR, CLR) starts from the input and reduces it to the start symbol. Yacc is bottom-up.

Q21. Why is left recursion bad for top-down but fine for yacc?

A recursive-descent parser loops forever on A -> A a. LR parsers handle left recursion well, and it even uses less stack than right recursion.

Q22. What is three-address code?

Intermediate code where every instruction has at most 3 addresses: x = y op z. Example: a = b + c * d becomes t1 = c * d ; t2 = b + t1 ; a = t2.

Q23. What is the difference between fgetc and fgets?

fgetc reads one character and returns int (EOF at the end). fgets reads one line into a buffer (keeping the '\n') and returns NULL at the end.

Q24. Why is ch declared as int and not char with fgetc?

fgetc returns EOF (-1) as an int. On systems where char is unsigned, a char can never equal -1, so the loop would never end.

Q25. What does ungetc do?

It pushes one character back into the input stream, so the next fgetc returns it again. It is used for 1-char look-ahead (e.g. '/' that is not followed by '/' or '*').

Q26. Difference between "w" and "a" mode?

"w" truncates (erases) the file, or creates it. "a" keeps the content and writes at the end.

Q27. What errors are detected by the lexical analyzer?

Illegal characters (@ or $ in C), unterminated strings or comments, invalid numbers like 12.3.4, and identifiers that are too long.

Q28. What is a regular expression? Give an identifier RE.

A pattern that describes a regular language. An identifier is letter(letter|digit)*, in lex: [a-zA-Z_][a-zA-Z0-9_]*.

Q29. What is the input buffering / sentinel concept?

The lexer reads input into buffer halves with two pointers (lexemeBegin, forward). A sentinel (eof) at the end of each half avoids checking the bounds on every character.

Q30. What is a context-free grammar?

G = (V, T, P, S): variables, terminals, productions of the form A -> alpha, and a start symbol. Yacc input is a CFG.
